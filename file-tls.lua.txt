local function report(feature, err)
	err = tostring(err)
	D.counts[feature] = (D.counts[feature] or 0) + 1
	D.lastFeature = feature
	D.lastErr = feature .. " :: " .. err
	if D.seen[feature] ~= err then
		D.seen[feature] = err
		if C.Verbose then
			warn("[Mu Delta Lite] " .. feature .. " -> " .. err .. " (x" .. D.counts[feature] .. ")")
		end
	end
end

-- run a function, report failures, never swallow silently
local function run(feature, fn, ...)
	local ok, err = pcall(fn, ...)
	if not ok then report(feature, err) end
	return ok, err
end

local function has(fnName)
	if D.feat[fnName] == nil then
		D.feat[fnName] = type(getfenv and rawget(getfenv(), fnName) or nil) == "function"
		if D.feat[fnName] == nil then D.feat[fnName] = false end
	end
	return D.feat[fnName]
end

local function tcount(t)
	local n = 0
	for _ in pairs(t) do n = n + 1 end
	return n
end

local function fire(...)
	if not CommF then return end
	local a = {...}
	-- NOTE: InvokeServer returns nil on server rejection; it does not throw.
	-- pcall can only catch Lua-side exceptions. Server refusal is invisible here.
	local ok, err = pcall(function() CommF:InvokeServer(unpack(a)) end)
	if not ok then report("remote:" .. tostring(a[1]), err) end
end

-- multi-source: Blox Fruits has moved this frame around between updates
local function isQuestOn()
	local pg = LP:FindFirstChild("PlayerGui")
	if not pg then return false end
	local main = pg:FindFirstChild("Main")
	local q = main and main:FindFirstChild("Quest")
	if q and q.Visible then return true end
	local tq = pg:FindFirstChild("TrackedQuestFrame")
	local f = tq and tq:FindFirstChild("Frame")
	if f and f.Visible then return true end
	return false
end

local activeTween, activeTarget
local function goTo(cf)
	local r = hrp()
	if not r or not cf then return end
	if activeTarget and (activeTarget.Position - cf.Position).Magnitude < 8 then return end
	if activeTween then pcall(function() activeTween:Cancel() end) end
	activeTarget = cf
	local dist = (r.Position - cf.Position).Magnitude
	if dist < 4 then return end
	local dur = math.clamp(dist / C.TweenSpeed, 0.08, 25)
	activeTween = TS:Create(r, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = cf})
	activeTween:Play()
end

local function hardTp(cf)
	local r = hrp()
	if not r then return end
	if activeTween then pcall(function() activeTween:Cancel() end) end
	activeTarget = nil
	r.CFrame = cf
end

-- device-agnostic click point (was hardcoded 1280x672 = desktop)
local function clickPoint()
	local cam = WS.CurrentCamera
	local vp = cam and cam.ViewportSize
	if vp and vp.X > 0 and vp.Y > 0 then
		return Vector2.new(vp.X / 2, vp.Y / 2)
	end
	return Vector2.new(1280, 672)
end

local function addQuest(lv, q, n, enemy)
	questMap[lv] = {
		quest = q, num = n, enemy = enemy,
		base = enemy:match("^(.-)%s*%[Lv%.") or enemy,
		tag = enemy:match("%[Lv%. %d+%]"),
	}
end

local function indexQuests()
	local module = RS:FindFirstChild("Quests")
	if not module then
		report("indexQuests", "ReplicatedStorage.Quests missing -> fallback table only")
	else
		local ok, Q = pcall(require, module)
		if not ok then
			report("indexQuests", "require(Quests) threw: " .. tostring(Q))
		elseif type(Q) ~= "table" then
			report("indexQuests", "Quests returned " .. type(Q) .. ", expected table")
		else
			local found = 0
			for name, nums in pairs(Q) do
				if name ~= "CitizenQuest" and type(nums) == "table" then
					for n, info in pairs(nums) do
						if type(info) == "table" and tonumber(info.LevelReq) and type(info.Task) == "table" then
							local enemy = next(info.Task)
							if enemy then addQuest(info.LevelReq, name, n, enemy); found = found + 1 end
						end
					end
				end
			end
			if found > 0 then D.questSource = "game module" end
			report("indexQuests", "module indexed " .. found .. " quests (informational)")
		end
	end

	local fromFallback = 0
	for _, e in ipairs(FB) do
		if not questMap[e[1]] then addQuest(e[1], e[2], e[3], e[4]); fromFallback = fromFallback + 1 end
	end
	if D.questSource == "none" then D.questSource = "fallback table" end

	D.quests = tcount(questMap)
	if D.quests == 0 then
		report("indexQuests", "ZERO quests indexed - auto farm level cannot work")
	end
end

local npcMap = {}
local function indexNpcs()
	local module = RS:FindFirstChild("GuideModule")
	if not module then
		report("indexNpcs", "ReplicatedStorage.GuideModule missing -> no NPC navigation")
		return
	end
	local ok, G = pcall(require, module)
	if not ok then
		report("indexNpcs", "require(GuideModule) threw: " .. tostring(G))
		return
	end
	if type(G) ~= "table" then
		report("indexNpcs", "GuideModule returned " .. type(G) .. ", expected table")
		return
	end
	local data = G.Data or G
	local list = data.NPCList
	if type(list) ~= "table" then
		report("indexNpcs", "GuideModule.Data.NPCList missing or not a table")
		return
	end
	for npc, info in pairs(list) do
		if typeof(npc) == "Instance" and type(info) == "table" and type(info.Levels) == "table" then
			local ok2, cf = pcall(function() return npc.CFrame end)
			if ok2 and typeof(cf) == "CFrame" then
				for _, lv in ipairs(info.Levels) do
					if type(lv) == "number" and not npcMap[lv] then npcMap[lv] = cf end
				end
			end
		end
	end
	D.npcs = tcount(npcMap)
	report("indexNpcs", "indexed " .. D.npcs .. " NPC anchor levels (informational)")
end

-- FIX: npcMap is keyed by NPC level; the old code looked it up with the QUEST level.
-- Use the highest NPC anchor <= player level instead of an exact match.
local function npcFor(lv)
	local bestLv, best = -1, nil
	for k, cf in pairs(npcMap) do
		if k <= lv and k > bestLv then bestLv, best = k, cf end
	end
	return best
end

local function questFor(lv)
	local bestLv, best = -1, nil
	for k, q in pairs(questMap) do
		if k <= lv and k > bestLv then bestLv, best = k, q end
	end
	return best, bestLv
end

--==================== enemies ====================
local function enemyFolder()
	return WS:FindFirstChild("Enemies")
end

local function findEnemy(full, base, tag)
	local folder = enemyFolder()
	local r = hrp()
	if not folder or not r then return nil end
	local ex, exD, lo, loD
	for _, v in pairs(folder:GetChildren()) do
		local h = v:FindFirstChildOfClass("Humanoid")
		if h and h.Health > 0 then
			local p = v:FindFirstChild("HumanoidRootPart")
			if p then
				local d = (p.Position - r.Position).Magnitude
				if v.Name == full then
					if not exD or d < exD then ex, exD = v, d end
				elseif base and v.Name:find(base, 1, true) and (not tag or v.Name:find(tag, 1, true)) then
					if not loD or d < loD then lo, loD = v, d end
				end
			end
		end
	end
	return ex or lo
end

local function findNearest(range)
	local folder = enemyFolder()
	local r = hrp()
	if not folder or not r then return nil end
	local best, bestD
	for _, v in pairs(folder:GetChildren()) do
		local h = v:FindFirstChildOfClass("Humanoid")
		if h and h.Health > 0 then
			local p = v:FindFirstChild("HumanoidRootPart")
			if p then
				local d = (p.Position - r.Position).Magnitude
				if d <= range and (not bestD or d < bestD) then best, bestD = v, d end
			end
		end
	end
	return best
end

local function bringMob(full, base, tag)
	local folder = enemyFolder()
	local r = hrp()
	if not folder or not r then return end
	if has("sethiddenproperty") then
		pcall(function()
			sethiddenproperty(LP, "SimulationRadius", math.huge)
			sethiddenproperty(LP, "MaxSimulationRadius", math.huge)
		end)
	end
	local pos = r.Position
	local n, slot = 0, 0
	for _, v in pairs(folder:GetChildren()) do
		if n >= 6 then break end
		if v.Name == full or (base and v.Name:find(base, 1, true) and (not tag or v.Name:find(tag, 1, true))) then
			local h = v:FindFirstChildOfClass("Humanoid")
			local p = v:FindFirstChild("HumanoidRootPart")
			if h and h.Health > 0 and p and (p.Position - pos).Magnitude < 400 then
				local off = Vector3.new(slot * 6, 3, 0)
				slot = slot + 1
				pcall(function()
					p.CanCollide = false
					p.Size = Vector3.new(30, 30, 30)
					h.PlatformStand = true
					p.CFrame = CFrame.new(pos + off)
				end)
				n = n + 1
			end
		end
	end
end

local CF
local cfOk, cfErr = pcall(function() CF = require(LP.PlayerScripts:WaitForChild("CombatFramework", 10)) end)
if not cfOk then report("combat", "CombatFramework require failed: " .. tostring(cfErr)) end

local function equipWeapon()
	if not alive() then return end
	local c = LP.Character
	local h = c and c:FindFirstChildOfClass("Humanoid")
	if not h then return end
	local name = C.Weapon
	-- FIX: "(none)" is a truthy placeholder; treat it as "no preference"
	if name and name ~= "(none)" then
		local t = LP.Backpack:FindFirstChild(name) or c:FindFirstChild(name)
		if t and t:IsA("Tool") and c:FindFirstChild(t.Name) == nil then
			h:EquipTool(t)
		end
		return
	end
	if c:FindFirstChildOfClass("Tool") then return end
	for _, t in pairs(LP.Backpack:GetChildren()) do
		if t:IsA("Tool") then h:EquipTool(t); break end
	end
end

local function attack()
	if not alive() then return end
	equipWeapon()
	if CF and C.FastAttack then
		local ok = pcall(function()
			local ac = CF.activeController
			if ac then
				ac.attacking = false
				ac.blocking = false
				ac.timeToNextAttack = 0
				ac.timeToNextBlock = 0
				ac.increment = 3
				ac.hitboxMagnitude = 120
			end
		end)
		if not ok then
			report("combat", "activeController fields rejected -> disabling fast attack")
			CF = nil
		end
	end
	pcall(function()
		VU:CaptureController()
		VU:Button1Down(clickPoint())
	end)
end

local function isFruit(inst)
	return inst:IsA("Tool") and inst.Name:find("Fruit") ~= nil and inst:FindFirstChild("Handle") ~= nil
end

local function notify(title, text)
	pcall(function()
		StarterGui:SetCore("SendNotification", {Title = title, Text = text, Duration = 6})
	end)
end

local function seedFruits()
	fruitSeen = {}
	for _, v in pairs(WS:GetChildren()) do
		if isFruit(v) then fruitSeen[v] = true end
	end
end

local function findGroundFruit(range)
	local r = hrp()
	if not r then return nil end
	local best, bestD
	for _, v in pairs(WS:GetChildren()) do
		if isFruit(v) then
			local d = (v.Handle.Position - r.Position).Magnitude
			if d <= range and (not bestD or d < bestD) then best, bestD = v, d end
		end
	end
	return best
end

local function heldFruit()
	local c = LP.Character
	if c then
		for _, t in pairs(c:GetChildren()) do
			if isFruit(t) then return t end
		end
	end
	local b = LP.Backpack
	if b then
		for _, t in pairs(b:GetChildren()) do
			if isFruit(t) then return t end
		end
	end
	return nil
end

local function loadSnipeList()
	local ok, res = pcall(function() return CommF:InvokeServer("GetFruits") end)
	if ok and type(res) == "table" and #res > 0 then
		snipeList = {}
		for _, f in ipairs(res) do
			if type(f) == "table" and f.Name then table.insert(snipeList, f.Name) end
		end
		if #snipeList == 0 then
			report("fruit", "GetFruits returned a table with no .Name fields")
		end
	else
		snipeList = {"Dough Fruit","Dragon Fruit","Leopard Fruit","Kitsune Fruit","Venom Fruit",
			"Shadow Fruit","Control Fruit","Spirit Fruit","Mammoth Fruit","T-Rex Fruit",
			"Gas Fruit","Yeti Fruit","Blizzard Fruit","Gravity Fruit","Portal Fruit"}
		report("fruit", "GetFruits unavailable -> using hardcoded snipe list")
	end
end

--==================== main loop ====================
local lastQuestFire, lastHaki, lastStats = 0, 0, 0

local function farmStep()
	local r = hrp()
	if not alive() or not r then return end

	if C.FarmNearest then
		local t = findNearest(C.Range)
		if t then
			local p = t:FindFirstChild("HumanoidRootPart")
			if p then
				if (p.Position - r.Position).Magnitude > 90 then
					goTo(p.CFrame * CFrame.new(0, 25, 0))
				end
				attack()
			end
		end
		return
	end

	if not C.FarmLevel then return end

	local lv = level()
	local q, qLv = questFor(lv)
	if not q then
		report("farm", "no quest entry for level " .. tostring(lv))
		return
	end

	if isQuestOn() then
		local t = findEnemy(q.enemy, q.base, q.tag)
		if t then
			local p = t:FindFirstChild("HumanoidRootPart")
			if p then
				local d = (p.Position - r.Position).Magnitude
				if d > 90 then
					goTo(p.CFrame * CFrame.new(0, 25, 0))
				elseif C.BringMob then
					bringMob(q.enemy, q.base, q.tag)
				end
				attack()
			end
		else
			local cf = npcFor(lv)
			if cf then goTo(cf) end
		end
	elseif C.AutoQuest then
		local now = os.clock()
		if now - lastQuestFire > 0.6 then
			lastQuestFire = now
			fire("StartQuest", q.quest, q.num)
			local cf = npcFor(lv)
			if cf and (cf.Position - r.Position).Magnitude > 45 then goTo(cf) end
		end
	end
end

local function fruitStep()
	local now = os.clock()

	if C.FruitSniper and now - lastSnipe > 0.5 and #snipeList > 0 then
		lastSnipe = now
		local name = snipeList[snipeIdx]
		snipeIdx = snipeIdx % #snipeList + 1
		if name then fire("PurchaseRawFruit", name, false) end
	end

	if C.AutoCollect then
		local f = findGroundFruit(4000)
		if f then goTo(f.Handle.CFrame) end
	end

	if C.AutoStore and now - lastStore > 1 then
		lastStore = now
		local t = heldFruit()
		if t and t.Parent == LP.Character then
			fire("StoreFruit", (t.Name:gsub(" Fruit$", "")), t)
		end
	end
end

task.spawn(function()
	while not stale() do
		D.loopTicks = D.loopTicks + 1

		run("farmStep", farmStep)
		run("fruitStep", fruitStep)

		local now = os.clock()

		if C.AutoHaki and now - lastHaki > 0.5 then
			lastHaki = now
			local c = LP.
