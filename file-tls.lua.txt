--==================== helpers ====================
local function hrp()
	local c = LP.Character
	return c and c:FindFirstChild("HumanoidRootPart")
end

local function alive()
	local h = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
	return h and h.Health > 0
end

local function level()
	local d = LP:FindFirstChild("Data")
	local l = d and d:FindFirstChild("Level")
	return (l and l.Value) or 1
end

local Remotes = RS:WaitForChild("Remotes", 30)
local CommF = Remotes and Remotes:WaitForChild("CommF_", 30)

local function fire(...)
	if not CommF then return end
	local a = {...}
	pcall(function() CommF:InvokeServer(unpack(a)) end)
end

local function isQuestOn()
	local ok, res = pcall(function()
		local main = LP.PlayerGui:FindFirstChild("Main")
		local q = main and main:FindFirstChild("Quest")
		return q and q.Visible
	end)
	return ok and res == true
end

-- tween with a single tracked target so loops don't fight each other
local activeTween, activeTarget
local function goTo(cf)
	local r = hrp()
	if not r or not cf then return end
	if activeTarget and (activeTarget.Position - cf.Position).Magnitude < 8 then return end
	if activeTween then pcall(function() activeTween:Cancel() end) end
	activeTarget = cf
	local dist = (r.Position - cf.Position).Magnitude
	if dist < 4 then return end
	local dur = math.clamp(dist / C.TweenSpeed, 0.08, 25)
	activeTween = TS:Create(r, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = cf})
	activeTween:Play()
end

local function hardTp(cf)
	local r = hrp()
	if not r then return end
	if activeTween then pcall(function() activeTween:Cancel() end) end
	activeTarget = nil
	r.CFrame = cf
end

local function addQuest(lv, q, n, enemy)
	questMap[lv] = {
		quest = q, num = n, enemy = enemy,
		base = enemy:match("^(.-)%s*%[Lv%.") or enemy,
		tag = enemy:match("%[Lv%. %d+%]"),
	}
end

local function indexQuests()
	local ok, Q = pcall(function() return require(RS.Quests) end)
	if ok and type(Q) == "table" then
		for name, nums in pairs(Q) do
			if name ~= "CitizenQuest" and type(nums) == "table" then
				for n, info in pairs(nums) do
					if type(info) == "table" and tonumber(info.LevelReq) and type(info.Task) == "table" then
						local enemy = next(info.Task)
						if enemy then addQuest(info.LevelReq, name, n, enemy) end
					end
				end
			end
		end
	end
	for _, e in ipairs(FB) do
		if not questMap[e[1]] then addQuest(e[1], e[2], e[3], e[4]) end
	end
end

local npcMap = {}
local function indexNpcs()
	local ok, G = pcall(function() return require(RS.GuideModule) end)
	if not ok or type(G) ~= "table" then return end
	local data = G.Data or G
	local list = data.NPCList
	if type(list) ~= "table" then return end
	for npc, info in pairs(list) do
		if typeof(npc) == "Instance" and type(info) == "table" and info.Levels then
			local ok2, cf = pcall(function() return npc.CFrame end)
			if ok2 and cf then
				for _, lv in ipairs(info.Levels) do
					if type(lv) == "number" and not npcMap[lv] then npcMap[lv] = cf end
				end
			end
		end
	end
end

local function questFor(lv)
	local bestLv, best = -1, nil
	for k, q in pairs(questMap) do
		if k <= lv and k > bestLv then bestLv, best = k, q end
	end
	return best, bestLv
end

--==================== enemies ====================
local function findEnemy(full, base, tag)
	local folder = WS:FindFirstChild("Enemies")
	local r = hrp()
	if not folder or not r then return nil end
	local ex, exD, lo, loD
	for _, v in pairs(folder:GetChildren()) do
		local h = v:FindFirstChildOfClass("Humanoid")
		if h and h.Health > 0 then
			local p = v:FindFirstChild("HumanoidRootPart")
			if p then
				local d = (p.Position - r.Position).Magnitude
				if v.Name == full then
					if not exD or d < exD then ex, exD = v, d end
				elseif v.Name:find(base, 1, true) and (not tag or v.Name:find(tag, 1, true)) then
					if not loD or d < loD then lo, loD = v, d end
				end
			end
		end
	end
	return ex or lo
end

local function findNearest(range)
	local folder = WS:FindFirstChild("Enemies")
	local r = hrp()
	if not folder or not r then return nil end
	local best, bestD
	for _, v in pairs(folder:GetChildren()) do
		local h = v:FindFirstChildOfClass("Humanoid")
		if h and h.Health > 0 then
			local p = v:FindFirstChild("HumanoidRootPart")
			if p then
				local d = (p.Position - r.Position).Magnitude
				if d <= range and (not bestD or d < bestD) then best, bestD = v, d end
			end
		end
	end
	return best
end

local function bringMob(full, base, tag)
	local folder = WS:FindFirstChild("Enemies")
	local r = hrp()
	if not folder or not r then return end
	pcall(function()
		sethiddenproperty(LP, "SimulationRadius", math.huge)
		sethiddenproperty(LP, "MaxSimulationRadius", math.huge)
	end)
	local pos = r.Position
	local n = 0
	for _, v in pairs(folder:GetChildren()) do
		if n >= 6 then break end
		if v.Name == full or (v.Name:find(base, 1, true) and (not tag or v.Name:find(tag, 1, true))) then
			local h = v:FindFirstChildOfClass("Humanoid")
			local p = v:FindFirstChild("HumanoidRootPart")
			if h and h.Health > 0 and p and (p.Position - pos).Magnitude < 400 then
				pcall(function()
					p.CanCollide = false
					p.Size = Vector3.new(30, 30, 30)
					h.PlatformStand = true
					p.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
				end)
				n = n + 1
			end
		end
	end
end

--==================== combat ====================
local CF
pcall(function() CF = require(LP.PlayerScripts:WaitForChild("CombatFramework", 10)) end)

local function equipWeapon()
	if not alive() then return end
	local c = LP.Character
	local h = c:FindFirstChildOfClass("Humanoid")
	if not h then return end
	local name = C.Weapon
	if name then
		local t = LP.Backpack:FindFirstChild(name)
		if t and t:IsA("Tool") then h:EquipTool(t) end
		return
	end
	if c:FindFirstChildOfClass("Tool") then return end
	for _, t in pairs(LP.Backpack:GetChildren()) do
		if t:IsA("Tool") then h:EquipTool(t) break end
	end
end

local function attack()
	if not alive() then return end
	equipWeapon()
	if CF and C.FastAttack then
		local ok = pcall(function()
			local ac = CF.activeController
			if ac then
				ac.attacking = false
				ac.blocking = false
				ac.timeToNextAttack = 0
				ac.timeToNextBlock = 0
				ac.increment = 3
				ac.hitboxMagnitude = 120
			end
		end)
		if not ok then CF = nil end
	end
	pcall(function()
		VU:CaptureController()
		VU:Button1Down(Vector2.new(1280, 672))
	end)
end

--==================== main loop ====================
local lastQuestFire, lastHaki, lastStats = 0, 0, 0

local function farmStep()
	local r = hrp()
	if not alive() or not r then return end

	if C.FarmNearest then
		local t = findNearest(C.Range)
		if t then
			local p = t:FindFirstChild("HumanoidRootPart")
			if p then
				if (p.Position - r.Position).Magnitude > 90 then
					goTo(p.CFrame * CFrame.new(0, 25, 0))
				end
				attack()
			end
		end
		return
	end

	if not C.FarmLevel then return end

	local q, qLv = questFor(level())
	if not q then return end

	if isQuestOn() then
		local t = findEnemy(q.enemy, q.base, q.tag)
		if t then
			local p = t:FindFirstChild("HumanoidRootPart")
			if p then
				local d = (p.Position - r.Position).Magnitude
				if d > 90 then
					goTo(p.CFrame * CFrame.new(0, 25, 0))
				elseif C.BringMob then
					bringMob(q.enemy, q.base, q.tag)
				end
				attack()
			end
		else
			local cf = qLv and npcMap[qLv]
			if cf then goTo(cf) end
		end
	elseif C.AutoQuest then
		local now = os.clock()
		if now - lastQuestFire > 0.6 then
			lastQuestFire = now
			fire("StartQuest", q.quest, q.num)
			local cf = qLv and npcMap[qLv]
			if cf and (cf.Position - r.Position).Magnitude > 45 then goTo(cf) end
		end
	end
end

task.spawn(function()
	while true do
		pcall(farmStep)

		local now = os.clock()

		if C.AutoHaki and now - lastHaki > 0.5 then
			lastHaki = now
			local c = LP.Character
			if c and not c:FindFirstChild("HasBuso") then fire("Buso") end
		end

		if now - lastStats > 0.7 then
			lastStats = now
			local d = LP:FindFirstChild("Data")
			local pts = d and d:FindFirstChild("Points")
			if pts and pts.Value >= C.Points then
				for stat, on in pairs(C.Stats) do
					if on then fire("AddPoint", stat, C.Points) end
				end
			end
		end

		task.wait(C.FarmLevel or C.FarmNearest and 0.12 or 0.2)
	end
end)

--==================== low graphics ====================
local function lowGraphics()
	pcall(function()
		local t = WS:FindFirstChildOfClass("Terrain")
		if t then
			t.WaterWaveSize = 0
			t.WaterWaveSpeed = 0
			t.WaterReflectance = 0
			t.WaterTransparency = 0
		end
		Lighting.GlobalShadows = false
		Lighting.FogEnd = 9e9
		Lighting.Brightness = 1
		pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level01 end)
		for _, v in pairs(WS:GetDescendants()) do
			if v:IsA("BasePart") or v:IsA("MeshPart") or v:IsA("UnionOperation") then
				v.Material = Enum.Material.Plastic
				v.Reflectance = 0
			elseif v:IsA("Decal") or v:IsA("Texture") then
				v.Transparency = 1
			elseif v:IsA("ParticleEmitter") or v:IsA("Trail") then
				v.Lifetime = NumberRange.new(0)
			elseif v:IsA("Explosion") then
				v.BlastPressure = 1
				v.BlastRadius = 1
			elseif v:IsA("Fire") or v:IsA("SpotLight") or v:IsA("Smoke") or v:IsA("Sparkles") then
				v.Enabled = false
			end
		end
		for _, e in pairs(Lighting:GetChildren()) do
			if e:IsA("BlurEffect") or e:IsA("SunRaysEffect") or e:IsA("ColorCorrectionEffect") or e:IsA("BloomEffect") or e:IsA("DepthOfFieldEffect") then
				e.Enabled = false
			end
		end
	end)
end

local function toggle(text, default, cb)
	local state = default
	local b = widget("")
	local function draw()
		b.Text = text .. "   [" .. (state and "ON" or "OFF") .. "]"
		b.TextColor3 = state and Color3.fromRGB(120, 235, 150) or Color3.fromRGB(235, 130, 130)
	end
	draw()
	b.MouseButton1Click:Connect(function()
		state = not state
		draw()
		pcall(cb, state)
	end)
	return b
end

local function cycle(text, options, idx, cb)
	local i = idx or 1
	local b = widget("")
	local function draw() b.Text = text .. ":  " .. tostring(options[i]) end
	draw()
	b.MouseButton1Click:Connect(function()
		i = i % #options + 1
		draw()
		pcall(cb, options[i])
	end)
	return b
end
